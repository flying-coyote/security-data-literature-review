% ==================== LEFT ARM: IDENTIFICATION ====================
\node (l-id) [box, fill=identification!30] at (0,-2.9) {
    \textbf{Records identified from databases}\\[2pt]
    OpenAlex (n = 354)\\
    dblp (n = 46)\\[3pt]
    \textbf{Total records identified: n = 400}\\[3pt]
    \footnotesize Search run 2026-07-13 (UTC). Boolean
    title-and-abstract filter at OpenAlex with a
    2018-01-01 date floor; dblp exposes no date
    parameter, so the date limit was enforced at
    screening for dblp records.
};

\node (l-dup) [sidebox, fill=identification!15] at (7.4,-2.9) {
    \textbf{Records removed before screening}\\[2pt]
    Duplicate records removed (n = 5)\\
    \quad by DOI (n = 1)\\
    \quad by normalized title (n = 4)\\[3pt]
    \footnotesize Cross-database overlap was a single
    record, so dblp contributed 45 records OpenAlex
    did not return.
};

% ==================== LEFT ARM: SCREENING ====================
\node (l-screen) [box, fill=screening!40] at (0,-6.7) {
    \textbf{Records screened (title / abstract)}\\[2pt]
    n = 395\\[3pt]
    \footnotesize 318 of 395 carried an abstract; the 45
    dblp-only records carry none and were judged on
    title and year.
};

\node (l-excl) [sidebox, fill=screening!25] at (7.4,-7.4) {
    \textbf{Records excluded (n = 355)}\\[3pt]
    \footnotesize
    E2 subject is not a security-telemetry data architecture: 137\\
    E1 ML/DL classifier evaluated on a streaming platform, the platform being plumbing rather than the object of study: 103\\
    I1 published before the 2018 date floor: 36\\
    E3 survey or review with no architecture-level evaluation of its own: 35\\
    E6 duplicate report of an already-included study: 13\\
    E4 not a substantive report (dataset or artifact descriptor): 11\\
    I2 no data-architecture subject: 10\\
    I3 no reported measurement, benchmark, or deployment: 8\\
    E5 not in English: 2
};

% ==================== LEFT ARM: ELIGIBILITY ====================
\node (l-elig) [box, fill=eligibility!50] at (0,-11.6) {
    \textbf{Reports assessed for eligibility}\\[2pt]
    n = 40\\[3pt]
    \footnotesize Eligibility was assessed at title and
    abstract level; no separate full-text retrieval
    round was run, so no reports were excluded at this
    stage (n = 0). Every decision and its criterion code
    is recorded per record in screen-batch-0..9.json.
};

% ==================== LEFT ARM: CRITICAL APPRAISAL ====================
% Screening was TOPICAL only. Nothing had yet asked whether the venues were real
% peer-reviewed venues, and a fifth of the yield turned out not to be.
\node (l-appr) [box, fill=eligibility!30] at (0,-15.2) {
    \textbf{Studies critically appraised}\\[2pt]
    n = 40\\[3pt]
    \footnotesize Venue identity resolved at the DOI; publisher, DOAJ, Scopus and Web of
    Science status established from primaries; predatory-list, delisting and hijacking
    checks run; then an independent LLM verification pass instructed to refute each
    proposed citation. Per-record verdicts and evidence: prisma-appraisal-2026-07-13.json.
};

\node (l-appr-excl) [sidebox, fill=eligibility!15] at (7.4,-15.9) {
    \textbf{Studies excluded at appraisal (n = 14)}\\[3pt]
    \footnotesize
    A1 predatory or compromised venue: 8\\
    \quad\emph{incl. 1 in a hijacked journal and 1 in a title delisted by Clarivate after
    an admitted paper-mill compromise}\\
    A2 not peer reviewed (2 preprints, 1 repository self-deposit): 3\\
    \quad\emph{incl. 1 preprint typeset with a counterfeit publisher masthead}\\
    A3 full text unobtainable, so uncitable: 1\\
    A4 citation refused by the independent LLM verification pass: 2
};

% ==================== LEFT ARM: INCLUDED ====================
\node (l-incl) [box, fill=included!50] at (0,-19.3) {
    \textbf{Studies included from databases}\\[2pt]
    n = 26\\[3pt]
    \footnotesize 11 Level A, 12 Level B, 3 Level C. The 26 resolve to 23 independent
    author groups. Full-text retrieval pass (2026-07-16): 19 of 26 read at full text,
    7 at abstract only, none unretrievable.
};

    analyst and industry reports: 14\\
    books: 8\\
    expert interviews and personal communications: 2\\
    first-party SDW lab measurements: 2\\
    documented no-primary stubs (retired or declined): 2\\

\node (r-elig) [box, fill=eligibility!50, text width=11.2cm, minimum width=11.8cm] at (18.6,-9.4) {

    n = 203\\[3pt]
    \footnotesize Reports not retrieved (n = 2): documented no-primary stubs kept as

% ==================== RIGHT ARM: INCLUDED ====================
\node (r-incl) [box, fill=included!50, text width=11.2cm, minimum width=11.8cm] at (18.6,-12.9) {

% ==================== CONVERGENCE ====================
\node (total) [box, fill=included!70, text width=14.4cm, minimum width=15cm] at (9.3,-24.2) {
    \textbf{\large Studies synthesised in this edition: n = 227}\\[3pt]
    201 tiered entries via the other-methods arm $+$ 26 via the database arm. The corpus
    carries 229 catalogued blocks, of which 227 hold an evidence tier and 2 are documented
    no-primary stubs.\\[3pt]
    \footnotesize Overlap between the two arms: n = 0. No database-included record matched a
    bibliography entry on DOI, on normalized title (exact, substring, or fuzzy), or on
    author and year, so measured recall of the curated corpus against the systematic search
    was 0 of 40 \emph{before} this incorporation. The two arms reach disjoint literatures, and
    that separation is a finding of the review rather than an artifact of the diagram: the
    curated arm holds the vendor and practitioner material a database cannot return, and the
    database arm holds the indexed literature the curation never reached. The 26 studies added
    here close the gap the retro-run exposed; the 14 the appraisal refused are recorded rather
    than quietly dropped, because a systematic review that reports only what it kept is not
    reporting its search.
};

% ==================== CAPTION ====================
\node [below=0.5cm of audit.south, font=\small, text width=24cm, align=left] (caption) {
    \textit{PRISMA 2020 two-arm flow. The database arm ran a boolean search of OpenAlex and a
    title search of dblp on 2026-07-13, identifying 400 records, of which 5 were duplicates,
    leaving 395 screened on title and abstract, 355 excluded against the recorded criteria, and
    40 included. The dominant exclusion was E2 (subject is not a security-telemetry data
    architecture, n = 137), followed by E1 (a machine-learning classifier evaluated on a
    streaming platform, where the platform is plumbing rather than the object of study, n = 103).
    The other-methods arm is the curated corpus of vendor engineering blogs, practitioner talks,
    standards documents, and technical books that a bibliographic database cannot return, since
    it is not journal or conference literature. The 40 database-included studies then passed
    through a critical-appraisal stage that the topical screening had not applied, and 14 did not
    survive it: eight sat in predatory or compromised venues, three were not peer reviewed, one
    could not be read, and two were refused by the independent LLM verification pass. The arms do not overlap,
    and each count above is reproducible from the JSON under methods/prisma-results/ and
    methods/prisma-appraisal-2026-07-13.json.}
};
