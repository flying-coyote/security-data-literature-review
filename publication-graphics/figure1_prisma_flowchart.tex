% PROVENANCE OF EVERY NUMBER IN THIS FIGURE:
%   methods/prisma-results/search-log.json      -- per-database raw counts, dedup breakdown
%   methods/prisma-results/records.json         -- 395 unique records carried into screening
%   methods/prisma-results/screen-batch-0..9.json -- 395 screening decisions + criterion codes
%   methods/prisma-results/reconciliation.json  -- corpus reconciliation, arm overlap, categories
% No count in this figure is estimated, rounded, or inferred.

\documentclass[tikz,border=5mm]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes,arrows,positioning,calc,fit}
\usepackage{xcolor}
\usepackage{amssymb}  % pdflatex-safe symbol set

% Define colors (unchanged four-stage scheme)
\definecolor{identification}{RGB}{144,202,249}  % Light blue
\definecolor{screening}{RGB}{255,224,178}        % Light orange
\definecolor{eligibility}{RGB}{255,245,157}      % Light yellow
\definecolor{included}{RGB}{165,214,167}         % Light green
\definecolor{bordercolor}{RGB}{33,33,33}         % Dark gray

% Define styles (same vocabulary as the other figures)
\tikzstyle{stage} = [rectangle, rounded corners, minimum width=13cm, minimum height=1.1cm,
                     text centered, text width=12.4cm, draw=bordercolor, line width=1.2pt,
                     font=\bfseries]
\tikzstyle{box} = [rectangle, minimum width=7.8cm, minimum height=1.6cm,
                   text width=7.2cm, draw=bordercolor, line width=0.8pt, font=\small,
                   align=left, inner sep=6pt]
\tikzstyle{sidebox} = [box, minimum width=5.6cm, text width=5.2cm]
\tikzstyle{arrow} = [thick,->,>=stealth,line width=1.5pt]
\tikzstyle{sidearrow} = [thick,->,>=stealth,line width=1.0pt]
\tikzstyle{label} = [font=\footnotesize\itshape]

\begin{document}

\begin{tikzpicture}

% ==================== ARM HEADERS ====================
\node (head-l) [stage, fill=identification] at (3.6,0)
    {Identification of studies via databases and registers};
\node (head-r) [stage, fill=identification] at (18.6,0)
    {Identification of studies via other methods};

% ==================== LEFT ARM: IDENTIFICATION ====================
\node (l-id) [box, fill=identification!30] at (0,-2.9) {
    \textbf{Records identified from databases}\\[2pt]
    OpenAlex (n = 354)\\
    dblp (n = 46)\\[3pt]
    \textbf{Total records identified: n = 400}\\[3pt]
    \footnotesize Search run 2026-07-13 (UTC). Boolean
    title-and-abstract filter at OpenAlex with a
    2018-01-01 date floor; dblp exposes no date
    parameter, so the date limit was enforced at
    screening for dblp records.
};

\node (l-dup) [sidebox, fill=identification!15] at (7.4,-2.9) {
    \textbf{Records removed before screening}\\[2pt]
    Duplicate records removed (n = 5)\\
    \quad by DOI (n = 1)\\
    \quad by normalized title (n = 4)\\[3pt]
    \footnotesize Cross-database overlap was a single
    record, so dblp contributed 45 records OpenAlex
    did not return.
};

% ==================== LEFT ARM: SCREENING ====================
\node (l-screen) [box, fill=screening!40] at (0,-6.7) {
    \textbf{Records screened (title / abstract)}\\[2pt]
    n = 395\\[3pt]
    \footnotesize 318 of 395 carried an abstract; the 45
    dblp-only records carry none and were judged on
    title and year.
};

\node (l-excl) [sidebox, fill=screening!25] at (7.4,-7.4) {
    \textbf{Records excluded (n = 355)}\\[3pt]
    \footnotesize
    E2 subject is not a security-telemetry data architecture: 137\\
    E1 ML/DL classifier evaluated on a streaming platform, the platform being plumbing rather than the object of study: 103\\
    I1 published before the 2018 date floor: 36\\
    E3 survey or review with no architecture-level evaluation of its own: 35\\
    E6 duplicate report of an already-included study: 13\\
    E4 not a substantive report (dataset or artifact descriptor): 11\\
    I2 no data-architecture subject: 10\\
    I3 no reported measurement, benchmark, or deployment: 8\\
    E5 not in English: 2
};

% ==================== LEFT ARM: ELIGIBILITY ====================
\node (l-elig) [box, fill=eligibility!50] at (0,-11.6) {
    \textbf{Reports assessed for eligibility}\\[2pt]
    n = 40\\[3pt]
    \footnotesize Eligibility was assessed at title and
    abstract level; no separate full-text retrieval
    round was run, so no reports were excluded at this
    stage (n = 0). Every decision and its criterion code
    is recorded per record in screen-batch-0..9.json.
};

% ==================== LEFT ARM: INCLUDED ====================
\node (l-incl) [box, fill=included!50] at (0,-15.0) {
    \textbf{Studies included from databases and registers}\\[2pt]
    n = 40
};

% ==================== RIGHT ARM: IDENTIFICATION ====================
\node (r-id) [box, fill=identification!30, text width=11.2cm, minimum width=11.8cm] at (18.6,-3.6) {
    \textbf{Records identified through the author's curation}\\[2pt]
    Curated corpus (MASTER-BIBLIOGRAPHY.md entry blocks): n = 195\\[3pt]
    \footnotesize
    vendor engineering blogs and product documentation: 63\\
    open-source project docs, specs, and repositories: 33\\
    practitioner talks, personal blogs, LinkedIn and Substack: 18\\
    standards, frameworks, and government publications: 17\\
    big-tech engineering blogs: 14\\
    analyst and industry reports: 12\\
    books: 8\\
    expert interviews and personal communications: 2\\
    first-party SDW lab measurements: 2\\
    documented no-primary stubs (retired or declined): 2\\
    peer-reviewed papers held in the corpus but not returned by the database search: 24
};

% ==================== RIGHT ARM: SCREENING / ELIGIBILITY ====================
\node (r-elig) [box, fill=eligibility!50, text width=11.2cm, minimum width=11.8cm] at (18.6,-9.4) {
    \textbf{Reports sought for retrieval and assessed for eligibility}\\[2pt]
    n = 195\\[3pt]
    \footnotesize Reports not retrieved (n = 2): documented no-primary stubs kept as
    the record of a claim whose source was retired or declined.
};

% ==================== RIGHT ARM: INCLUDED ====================
\node (r-incl) [box, fill=included!50, text width=11.2cm, minimum width=11.8cm] at (18.6,-12.9) {
    \textbf{Studies included via other methods}\\[2pt]
    n = 193 (each carrying an evidence tier)
};

% ==================== CONVERGENCE ====================
\node (total) [box, fill=included!70, text width=14.4cm, minimum width=15cm] at (9.3,-18.3) {
    \textbf{\large Total studies included in the review: n = 233}\\[3pt]
    40 from databases and registers $+$ 193 from the curated corpus.\\[3pt]
    \footnotesize Overlap between the two arms: n = 0. No database-included record matched a
    bibliography entry on DOI, on normalized title (exact, substring, or fuzzy), or on
    author and year, so corpus recall against the systematic search is 0 of 40. The two arms
    reach disjoint literatures, and that separation is a finding of the review rather than an
    artifact of the diagram.
};

% ==================== ARROWS ====================
\draw [arrow] (l-id.south) -- (l-screen.north);
\draw [arrow] (l-screen.south) -- (l-elig.north);

\draw [sidearrow] (l-id.east) -- (l-dup.west);

% ==================== AUDIT BANNER ====================
\node (audit) [box, fill=white, text width=14.4cm, minimum width=15cm, draw=bordercolor,
               line width=1.0pt] at (9.3,-21.6) {
    \textbf{Audit banner (2026-06-14).} The aggregate Level-A self-grade is WITHDRAWN and no
    aggregate percentage is claimed pending re-verification; per-source evidence levels remain
    provisional. Of the 195 bibliography blocks, 193 carry an evidence level.
};

% ==================== TITLE ====================
\node [above=0.6cm of head-l.north west, anchor=west, font=\LARGE\bfseries] (title) {
    Figure 1: PRISMA 2020 flow diagram
};

% ==================== CAPTION ====================
\node [below=0.5cm of audit.south, font=\small, text width=24cm, align=left] (caption) {
    \textit{PRISMA 2020 two-arm flow. The database arm ran a boolean search of OpenAlex and a
    title search of dblp on 2026-07-13, identifying 400 records, of which 5 were duplicates,
    leaving 395 screened on title and abstract, 355 excluded against the recorded criteria, and
    40 included. The dominant exclusion was E2 (subject is not a security-telemetry data
    architecture, n = 137), followed by E1 (a machine-learning classifier evaluated on a
    streaming platform, where the platform is plumbing rather than the object of study, n = 103).
    The other-methods arm is the curated corpus of vendor engineering blogs, practitioner talks,
    standards documents, and technical books that a bibliographic database cannot return, since
    it is not journal or conference literature. The arms do not overlap, and each count above is
    reproducible from the JSON under methods/prisma-results/.}
};
